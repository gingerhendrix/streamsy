```ts
  type PublicErrorTag,
} from "@streamsy/serve/contract";
```

`@streamsy/serve/contract` holds the header names, the JSON encoding, and the error bodies that a Serve route and its clients share. See [Browser clients and CORS](/docs/http/serve#browser-clients-and-cors).

## `canonicalJson`

```ts
// ---cut-start---
import type * as Real from "@streamsy/serve/contract";
import type { CanonicalValue } from "@streamsy/serve/contract";
import { same } from "./reference-check.ts";
// ---cut-end---
declare function canonicalJson(value: CanonicalValue): string;
// ---cut-after---
same<typeof canonicalJson, typeof Real.canonicalJson>(true);
```

`canonicalJson` encodes a value as JSON with the keys of every object sorted and the order of every array kept. A `null` value remains `null`. A non-finite number throws a `TypeError` when the encoder visits it.

Pass ordinary objects. An object with a null prototype fails the `instanceof Object` test, so `canonicalJson` encodes it with `JSON.stringify`: it does not sort its keys, and it writes a non-finite number in it as `null`.

The type excludes `undefined`. When an object member is `undefined` at run time, `canonicalJson` leaves it out.

## `contractFingerprint`

```ts
// ---cut-start---
import type * as Real from "@streamsy/serve/contract";
import type { CanonicalValue } from "@streamsy/serve/contract";
import { same } from "./reference-check.ts";
// ---cut-end---
declare function contractFingerprint(value: CanonicalValue): string;
// ---cut-after---
same<typeof contractFingerprint, typeof Real.contractFingerprint>(true);
```

`contractFingerprint` returns eight lowercase hexadecimal characters: an FNV-1a hash of the UTF-16 code units of `canonicalJson(value)`. Two ordinary objects that differ only in property order have the same fingerprint. The fingerprint detects a change. It is not a cryptographic hash.

## `valueEtag`

```ts
// ---cut-start---
import type * as Real from "@streamsy/serve/contract";
import { same } from "./reference-check.ts";
// ---cut-end---
declare function valueEtag(canonical: string): string;
// ---cut-after---
same<typeof valueEtag, typeof Real.valueEtag>(true);
```

`valueEtag` returns a strong ETag: sixteen lowercase hexadecimal characters inside double quotes. It concatenates two FNV-1a hashes of `canonical` with different seeds. To tag a JSON value, pass the result of `canonicalJson`. Like the fingerprint, the ETag is not a cryptographic hash.

## `decodePublicError`

```ts
// ---cut-start---
import type * as Real from "@streamsy/serve/contract";
import type { PublicError } from "@streamsy/serve/contract";
import { same } from "./reference-check.ts";
// ---cut-end---
declare function decodePublicError(value: unknown): PublicError;
// ---cut-after---
same<typeof decodePublicError, typeof Real.decodePublicError>(true);
```

`decodePublicError` validates an untrusted error object and returns a new `PublicError` without the unknown fields. It throws a `TypeError` when the body is not valid or its tag is unknown.

`decodePublicError` reads each required field only as an own data property, so it never calls a getter. It rejects an inherited field, an accessor, a boxed string, an array, an object with a null prototype, and `null`.

## `CanonicalValue`

```ts
// ---cut-start---
import type * as Real from "@streamsy/serve/contract";
import { same } from "./reference-check.ts";
// ---cut-end---
type CanonicalValue =
  | string
  | number
  | boolean
  | null
  | readonly CanonicalValue[]
  | { readonly [key: string]: CanonicalValue };
// ---cut-after---
same<CanonicalValue, Real.CanonicalValue>(true);
```

`CanonicalValue` is the type of a JSON value: a string, a number, a boolean, `null`, a readonly array, or an object with string keys. The compiler accepts `NaN` and `Infinity` as a `number`, but `canonicalJson` throws on them.

## `ReplayRecovery`

```ts

import { same } from "./reference-check.ts";
// ---cut-end---
type ReplayRecovery = "replay-from-start";
// ---cut-after---
same<ReplayRecovery, Real.ReplayRecovery>(true);
```

`ReplayRecovery` is the `recovery` value of the stream and state errors. On `"replay-from-start"`, a state client clears its local rows and reads again from offset `-1`.

## `ResumeRejectedReason`

```ts
// ---cut-start---
import type * as Real from "@streamsy/serve/contract";
import { same } from "./reference-check.ts";
// ---cut-end---
type ResumeRejectedReason = "invalid-offset" | "history-unavailable" | "contract-changed";
// ---cut-after---
same<ResumeRejectedReason, Real.ResumeRejectedReason>(true);
```

`ResumeRejectedReason` is the `reason` of a `ResumeRejected` error.

| Reason                | Meaning                                                              |
| --------------------- | -------------------------------------------------------------------- |
| `invalid-offset`      | The offset is malformed, or the read rejects it                      |
| `history-unavailable` | The stream to resume does not exist or was deleted                   |
| `contract-changed`    | The contract fingerprint of the request differs from the current one |

## `PublicError`

```ts
// ---cut-start---
import type * as Real from "@streamsy/serve/contract";
import type { ReplayRecovery, ResumeRejectedReason } from "@streamsy/serve/contract";
import { same } from "./reference-check.ts";
// ---cut-end---
type PublicError =
  | {
      readonly _tag: "InvalidParams";
      readonly route: string;
      readonly parameter: string;
      readonly detail: string;
    }
  | {
      readonly _tag: "ProtocolVersionUnsupported";
      readonly route: string;
      readonly supported: 1;
      readonly received: string;
      readonly recovery: ReplayRecovery;
    }
  | {
      readonly _tag: "ResumeRejected";
      readonly route: string;
      readonly reason: ResumeRejectedReason;
      readonly recovery: ReplayRecovery;
    }
  | { readonly _tag: "ContractChanged"; readonly route: string; readonly recovery: "refetch" }
  | {
      readonly _tag: "TransportUnavailable" | "WireEncodeFailed";
      readonly route: string;
      readonly detail: string;
    };
// ---cut-after---
same<PublicError, Real.PublicError>(true);
```

`PublicError` is the type of the JSON body of a Serve error. `route` is the matched route pattern. In a body from Serve, `parameter` is always `"params"`. `detail` describes the failure of `InvalidParams`, `TransportUnavailable`, or `WireEncodeFailed`.

`recovery` tells the client what to do next: `"replay-from-start"` for the stream and state errors, and `"refetch"` for `ContractChanged`. For the fields and the HTTP status of each error, see [Serve errors](/docs/reference/serve/serve-errors).

## `PublicErrorTag`

```ts

```

`PublicErrorTag` is the union of the `_tag` values of `PublicError`, which are the entries of `PUBLIC_ERROR_TAGS`.
