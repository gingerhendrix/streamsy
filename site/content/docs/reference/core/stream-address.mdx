---
title: StreamAddress
description: Reference for the StreamAddress namespace in @streamsy/core. An address names one stream and holds its codec.
---

```ts
import { StreamAddress, type StreamId } from "@streamsy/core";
import type { Schema } from "effect";
```

An address names one stream and gives the codec of its items. Making an address gets no service and no resource. A stream class makes its addresses with `at(params)`, so most code does not call these functions. See [Templates and addresses](/docs/streams/declare#templates-and-addresses).

The State types follow the [Durable State protocol](https://durablestreams.com/durable-state). For the change helpers that most writers use, see [`DurableState`](/docs/reference/core/durable-state).

## Addresses

### `StreamAddress.json`

```ts
// ---cut-start---
import type { Schema } from "effect";
import { StreamAddress } from "@streamsy/core";
import { same } from "./reference-check.ts";
// ---cut-end---
declare function json<A, I, RD, RE>(
  id: string,
  options: { readonly schema: Schema.Codec<A, I, RD, RE> },
): StreamAddress.StreamAddress<A, RD, RE>;
// ---cut-after---
same<typeof json, typeof StreamAddress.json>(true);
```

The address of a JSON stream. The content type is `application/json`. The codec encodes each item to a JSON string with `schema`.

### `StreamAddress.state`

```ts
// ---cut-start---
import { StreamAddress } from "@streamsy/core";
import { same } from "./reference-check.ts";
// ---cut-end---
declare function state<const C extends StreamAddress.Collections>(
  id: string,
  options: { readonly collections: C & StreamAddress.ValidCollections<C> },
): StreamAddress.StateAddress<C>;
// ---cut-after---
same<typeof state, typeof StreamAddress.state>(true);
```

The address of a Durable State stream. It is a JSON address. Each key of `collections` is the `type` of one collection on the wire. The codec reads and writes the changes of every collection and the three State controls.

### `StreamAddress.StreamAddress`

```ts
// ---cut-start---
import type { Schema } from "effect";
import { StreamAddress, type StreamId } from "@streamsy/core";
import { same } from "./reference-check.ts";
// ---cut-end---
interface StreamAddress<A, RD = never, RE = never> {
  readonly _tag: "Json" | "Bytes";
  readonly id: StreamId;
  readonly contentType: string;
  readonly codec: Schema.Codec<A, string | Uint8Array, RD, RE>;
}
// ---cut-after---
interface Decoding {
  readonly decoding: unique symbol;
}
same<StreamAddress<string>, StreamAddress.StreamAddress<string>>(true);
same<StreamAddress<string, Decoding>, StreamAddress.StreamAddress<string, Decoding>>(true);
```

| Field         | Meaning                                                                                          |
| ------------- | ------------------------------------------------------------------------------------------------ |
| `_tag`        | `Json` when each item is one message, `Bytes` when the items of one append join into one message |
| `id`          | The stream id                                                                                    |
| `contentType` | The content type that `create` sends                                                             |
| `codec`       | Encodes an item to the wire form, and decodes it back                                            |

`RD` holds the services that the codec needs to decode, and `RE` the services that it needs to encode.

### `StreamAddress.StateAddress`

```ts
// ---cut-start---
import { StreamAddress } from "@streamsy/core";
import { same } from "./reference-check.ts";
// ---cut-end---
interface StateAddress<
  C extends StreamAddress.Collections,
  RD = StreamAddress.CollectionsDecodingServices<C>,
  RE = StreamAddress.CollectionsEncodingServices<C>,
> extends StreamAddress.StreamAddress<StreamAddress.CollectionsEvent<C>, RD, RE> {
  readonly collections: C;
}
// ---cut-after---
type C = StreamAddress.Collections;
same<StateAddress<C>, StreamAddress.StateAddress<C>>(true);
```

A JSON address that also keeps its collections. `DurableState.changes` reads the key field of each collection from `collections`.

## Collections

### `StreamAddress.ValidCollections`

```ts
// ---cut-start---
import { Schema } from "effect";
import { StreamAddress } from "@streamsy/core";
import { same } from "./reference-check.ts";
// ---cut-end---
type ValidCollections<C extends StreamAddress.Collections> = {
  readonly [K in keyof C]: C[K]["key"] extends StreamAddress.StateKey<C[K]["schema"]["Type"]>
    ? C[K]
    : { readonly key: StreamAddress.StateKey<C[K]["schema"]["Type"]> };
};
// ---cut-after---
const Issue = Schema.Struct({ id: Schema.String, done: Schema.Boolean });
type Good = { readonly issue: { readonly schema: typeof Issue; readonly key: "id" } };
type Bad = { readonly issue: { readonly schema: typeof Issue; readonly key: "done" } };
same<ValidCollections<Good>, StreamAddress.ValidCollections<Good>>(true);
same<ValidCollections<Bad>, StreamAddress.ValidCollections<Bad>>(true);
```

Checks that each `key` names a `string` or `number` field of its own collection. A wrong key keeps the other collections as they are. The compiler then reports the error on the `key` of the wrong collection.

### `StreamAddress.CollectionsEncodedChange`

```ts
// ---cut-start---
import { Schema } from "effect";
import { StreamAddress } from "@streamsy/core";
import { same } from "./reference-check.ts";
// ---cut-end---
type CollectionsEncodedChange<C extends StreamAddress.Collections> = {
  [K in keyof C & string]: StateEncodedChange<C[K]["schema"]["Encoded"], K>;
}[keyof C & string];

// StateEncodedChange has no exported name. It is the wire form of one change.
type StateEncodedChange<A, Type extends string = string> =
  | (Omit<StreamAddress.StateUpsert<A, Type>, "headers"> & {
      readonly headers: Omit<StreamAddress.StateHeaders, "operation"> & {
        readonly operation: "insert" | "update" | "upsert";
      };
    })
  | StreamAddress.StateDelete<A, Type>;
// ---cut-after---
const Issue = Schema.Struct({ id: Schema.String, rank: Schema.NumberFromString });
type C = { readonly issue: { readonly schema: typeof Issue; readonly key: "id" } };
same<CollectionsEncodedChange<C>, StreamAddress.CollectionsEncodedChange<C>>(true);
```

The wire form of the changes of every collection. An upsert on the wire can have the operation `insert`, `update` or `upsert`. A read decodes all three as `upsert`.

## Changes and controls

### `StreamAddress.isChange`

```ts
// ---cut-start---
import { StreamAddress } from "@streamsy/core";
import { same } from "./reference-check.ts";
// ---cut-end---
declare function isChange<E extends StreamAddress.StateEvent<unknown>>(
  event: E,
): event is Exclude<E, StreamAddress.StateControl>;
// ---cut-after---
same<typeof isChange, typeof StreamAddress.isChange>(true);
```

`true` for an upsert or a delete. Use it before you read the `type` or the `key` of a State item.

### `StreamAddress.isControl`

```ts
// ---cut-start---

// ---cut-after---

```

`true` for a control message. A control has `headers.control`, and a change has `headers.operation`.

### `StreamAddress.snapshotStart`

```ts
// ---cut-start---
import { StreamAddress } from "@streamsy/core";
import { same } from "./reference-check.ts";
// ---cut-end---
declare function snapshotStart(options?: { readonly offset?: string }): StreamAddress.StateControl;
// ---cut-after---
same<typeof snapshotStart, typeof StreamAddress.snapshotStart>(true);
```

A `snapshot-start` control. The changes from here to the next `snapshot-end` are a complete snapshot.

### `StreamAddress.snapshotEnd`

```ts
// ---cut-start---
import { StreamAddress } from "@streamsy/core";
import { same } from "./reference-check.ts";
// ---cut-end---
declare function snapshotEnd(options?: { readonly offset?: string }): StreamAddress.StateControl;
// ---cut-after---
same<typeof snapshotEnd, typeof StreamAddress.snapshotEnd>(true);
```

A `snapshot-end` control. It closes the snapshot that `snapshot-start` opened.

The three control functions leave out `headers.offset` when you do not give `offset`.

```ts
import { StreamAddress } from "@streamsy/core";

const controls = [StreamAddress.snapshotStart(), StreamAddress.snapshotEnd({ offset: "42" })];
```

`controls[0].headers` is `{ control: "snapshot-start" }`.

### `StreamAddress.stateChange`

```ts
// ---cut-start---
import type { Schema } from "effect";
import { StreamAddress } from "@streamsy/core";
import { same } from "./reference-check.ts";
// ---cut-end---
declare function stateChange<A, I, RD, RE, const Type extends string>(options: {
  readonly schema: Schema.Codec<A, I, RD, RE>;
  readonly type: Type;
}): Schema.Codec<StreamAddress.StateChange<A, Type>, StateEncodedChange<I, Type>, RD, RE>;

// StateEncodedChange has no exported name. It is the wire form of one change.
type StateEncodedChange<A, Type extends string = string> =
  | (Omit<StreamAddress.StateUpsert<A, Type>, "headers"> & {
      readonly headers: Omit<StreamAddress.StateHeaders, "operation"> & {
        readonly operation: "insert" | "update" | "upsert";
      };
    })
  | StreamAddress.StateDelete<A, Type>;
// ---cut-after---
same<typeof stateChange, typeof StreamAddress.stateChange>(true);
```

The codec for the upserts and the deletes of one collection. `state` makes one of these for each collection. Leave out an optional header. Do not set it to `undefined`.
